import { Card, CardGrid } from '@astrojs/starlight/components';

:::note[Status]
The collector records a dataset; nothing scales yet. Early burst detection is measured offline against the HPA; the detector and the scaler are not validated.
:::

<CardGrid>
	<Card title="Sees requests before CPU" icon="rocket">
		A web server answers each request with about two `sendto` calls. Their rate is the request rate, seen from inside the node, within one 2 s poll of a burst.
	</Card>
	<Card title="eBPF, /proc and cgroups" icon="puzzle">
		eBPF for events no file keeps: syscalls, waits for a CPU, calls into the GPU driver. Files for the counters the kernel already keeps.
	</Card>
	<Card title="Knows what a task runs" icon="list-format">
		Framework from the libraries a process loaded: PyTorch, TensorFlow, JAX, vLLM, Ray, Spark and more, from a JSON catalog.
	</Card>
	<Card title="A dataset per node" icon="document">
		One JSON line per process per poll, rotated hourly into zstd segments, joined to experiments by container id.
	</Card>
</CardGrid>
